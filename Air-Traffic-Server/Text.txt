
import asyncio
import json
from pathlib import Path

from fastapi import FastAPI, WebSocket, WebSocketDisconnect
from fastapi.responses import FileResponse
from fastapi.staticfiles import StaticFiles

app = FastAPI()

BASE_DIR = Path(__file__).resolve().parent
STATIC_DIR = BASE_DIR / "static"

app.mount("/static", StaticFiles(directory=STATIC_DIR), name="static")

planes = {}
clients = set()
next_id = 1
state_lock = asyncio.Lock()


@app.get("/")
async def tower():
    return FileResponse(STATIC_DIR / "index.html")


async def broadcast_state():
    message = json.dumps({
        "type": "state",
        "planes": list(planes.values()),
    })

    disconnected = set()

    for client in list(clients):
        try:
            await client.send_text(message)
        except Exception:
            disconnected.add(client)

    clients.difference_update(disconnected)


@app.websocket("/ws")
async def websocket_endpoint(websocket: WebSocket):
    global next_id

    await websocket.accept()
    clients.add(websocket)

    async with state_lock:
        plane_id = next_id
        next_id += 1

        planes[plane_id] = {
            "id": plane_id,
            "callsign": f"FLT-{plane_id:03}",
            "x": 50 + (plane_id * 73) % 700,
            "y": 50 + (plane_id * 47) % 440,
            "vx": 20,
            "vy": 15,
            "status": "AIRBORNE",
        }

    await broadcast_state()

    try:
        while True:
            message = await websocket.receive_text()

            try:
                signal = json.loads(message)
            except json.JSONDecodeError:
                await websocket.send_json({
                    "type": "error",
                    "message": "Invalid JSON",
                })
                continue

            # For now, only allow simple landing requests.
            if signal.get("type") == "LAND_REQUEST":
                async with state_lock:
                    if plane_id in planes:
                        planes[plane_id]["status"] = "LAND_REQUESTED"

                await broadcast_state()

    except WebSocketDisconnect:
        pass
    finally:
        clients.discard(websocket)

        async with state_lock:
            planes.pop(plane_id, None)

        await broadcast_state()


async def simulation_loop():
    while True:
        await asyncio.sleep(0.1)

        async with state_lock:
            for plane in planes.values():
                plane["x"] += plane["vx"] * 0.1
                plane["y"] += plane["vy"] * 0.1

                if plane["x"] < 15 or plane["x"] > 785:
                    plane["vx"] *= -1
                if plane["y"] < 15 or plane["y"] > 525:
                    plane["vy"] *= -1

                plane["x"] = max(15, min(785, plane["x"]))
                plane["y"] = max(15, min(525, plane["y"]))

        await broadcast_state()


@app.on_event("startup")
async def startup():
    asyncio.create_task(simulation_loop())
